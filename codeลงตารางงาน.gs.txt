  function formatNumber(num) { 
      if(num === null || num === undefined || num === '' || num === '-') return '-'; 
      let v = typeof num === 'string' ? parseFloat(num.replace(/,/g, '')) : parseFloat(num); 
      if (isNaN(v)) return '-'; 
      return v.toLocaleString('th-TH', {minimumFractionDigits: 0, maximumFractionDigits: 2}); 
  }
  
  function parseTimeStr(val) { 
      if (!val) return null; 
      const str = String(val).trim(); 
      const parts = str.replace('.', ':').split(':'); 
      const h = parseInt(parts[0]); 
      const m = parseInt(parts[1]) || 0; 
      if (isNaN(h)) return null; 
      return { h, m, val: h + m/60 }; 
  }
  
  function getScheduleCellStyle(h, workStart, workEnd, breakStart, breakEnd) {
      const points = new Set([0, 1]);
      const addPoint = (t) => { const p = t - h; if (p > 0 && p < 1) points.add(p); };
      addPoint(workStart); addPoint(workEnd);
      if (breakStart !== -1) { addPoint(breakStart); addPoint(breakEnd); }
      const sorted = Array.from(points).sort((a, b) => a - b);
      const gradient = [];
      for (let i = 0; i < sorted.length - 1; i++) {
          const sPct = sorted[i] * 100; const ePct = sorted[i+1] * 100;
          const mid = h + (sorted[i] + sorted[i+1]) / 2;
          let color = 'transparent';
          let isBreak = (breakStart !== -1 && mid >= breakStart && mid < breakEnd);
          let isWork = (mid >= workStart && mid < workEnd);
          if (isBreak) color = '#ffc107'; else if (isWork) color = '#28a745';
          gradient.push(`${color} ${sPct}%`); gradient.push(`${color} ${ePct}%`);
      }
      if (gradient.length === 0) return '';
      const colors = new Set(gradient.map(p => p.split(' ')[0]));
      if (colors.size === 1) { const c = colors.values().next().value; if(c==='transparent') return ''; return `background-color: ${c};`; }
      return `background: linear-gradient(to right, ${gradient.join(', ')});`;
  }

  function initExpWeek() {
      let d = new Date();
      let day = d.getDay(); // 0=Sun, 1=Mon, ..., 6=Sat
      let diff = d.getDate() - day + (day === 0 ? -6 : 1); // หาวันจันทร์
      expWeekStartDate = new Date(d.setDate(diff));
      expWeekStartDate.setHours(0,0,0,0);
      updateExpWeekDisplay();
  }

  function changeExpWeek(weeks) {
      expWeekStartDate.setDate(expWeekStartDate.getDate() + (weeks * 7));
      updateExpWeekDisplay();
      if(document.getElementById('expBranchSelect').value) {
          loadExpEmployees(); 
      }
  }

  function updateExpWeekDisplay() {
      let end = new Date(expWeekStartDate);
      end.setDate(end.getDate() + 6); // วันอาทิตย์
      const formatObj = { day: 'numeric', month: 'short', year: 'numeric' };
      const str1 = expWeekStartDate.toLocaleDateString('th-TH', formatObj);
      const str2 = end.toLocaleDateString('th-TH', formatObj);
      document.getElementById('expWeekDisplay').innerText = `${str1} - ${str2}`;
  }

  function updateExpDashboard(s) {
      if(s && s.success) {
          let dailyTgt = parseFloat(String(s.dailyTarget).replace(/,/g, '')) || 0;

      }
  }

  async function loadExpEmployees() {
      const b = document.getElementById('expBranchSelect').value; if(!b) return;
      document.getElementById('expTableBody').innerHTML = `<tr><td colspan="8" class="text-center py-5"><div class="spinner-border text-primary"></div><br>กำลังดึงข้อมูลพนักงานและตารางงาน...</td></tr>`;
      
      google.script.run.withSuccessHandler(updateExpDashboard).getBranchStats(b);

      expDataMap = {}; 
      expModifiedKeys.clear(); 
      currentLoadedEmployees = [];

      try {
          let emps = await new Promise((resolve, reject) => {
              google.script.run.withSuccessHandler(resolve).withFailureHandler(reject).getEmployeesByBranch(b);
          });
          if(emps) { emps.sort((a, b) => getPositionPriority(a.position) - getPositionPriority(b.position)); }
          currentLoadedEmployees = emps || [];

          let curDate = new Date(expWeekStartDate);
          let fetchPromises = [];
          for(let i=0; i<7; i++) {
              let dStr = formatDateLocal(curDate);
              fetchPromises.push(new Promise((resolve) => {
                  google.script.run
                      .withSuccessHandler(res => resolve({date: dStr, res: res}))
                      .withFailureHandler(err => resolve({date: dStr, res: {success: false, message: err.message}}))
                      .getHistoryData(dStr, b);
              }));
              curDate.setDate(curDate.getDate() + 1);
          }

          const weeklyResults = await Promise.all(fetchPromises);
          
          weeklyResults.forEach(item => {
              if (item.res && item.res.success && item.res.data) {
                  item.res.data.forEach(row => {
                      let emp = currentLoadedEmployees.find(e => e.name === row.name);
                      if (emp) {
                          let key = `${emp.hrCode}_${item.date}`;
                          if(row.otherNote === "ล้างข้อมูล") return;

                          let payload = {
                              workDate: item.date, hrCode: emp.hrCode, name: emp.name, branch: b, position: emp.position, empType: emp.empType,
                              checkIn: row.checkIn !== '-' ? row.checkIn : '', checkOut: row.checkOut !== '-' ? row.checkOut : '', breakTime: row.breakTime !== '-' ? row.breakTime : '', 
                              breakTimeRange: row.breakTimeRange !== '-' ? row.breakTimeRange : '', ot: row.ot !== '-' ? row.ot : '0', otAccumulated: row.otAccumulated !== '-' ? row.otAccumulated : '0', 
                              wage: parseFloat(String(row.wage).replace(/,/g, '')) || 0, isStop: row.status === 'หยุด',
                              leaveNote: row.note !== '-' ? row.note : '', unpaidLeave: row.unpaidLeave !== '-' ? row.unpaidLeave : '', hourlyLeave: row.hourlyLeave !== '-' ? row.hourlyLeave : '', 
                              otherNote: row.otherNote !== '-' ? row.otherNote : '', workStation: row.workStation !== '-' ? row.workStation : '', useAccumulatedHours: row.useAccumulatedHours !== '-' ? row.useAccumulatedHours : ''
                          };
                          expDataMap[key] = payload;
                      }
                  });
              }
          });

          renderExpGrid();
          document.getElementById('expSummaryPanel').style.display = 'flex';
      } catch(e) {
          document.getElementById('expTableBody').innerHTML = `<tr><td colspan="8" class="text-danger text-center py-4">Error: ${e.message}</td></tr>`;
      }
  }

  function renderExpGrid() {
      const thead = document.getElementById('expTableHead');
      const tbody = document.getElementById('expTableBody');
      if(currentLoadedEmployees.length === 0) {
          tbody.innerHTML = `<tr><td colspan="8" class="text-center py-4">ไม่พบพนักงานในสาขานี้</td></tr>`; return;
      }

      let curDate = new Date(expWeekStartDate);
      let dateStrings = []; 
      for(let i=0; i<7; i++) {
          dateStrings.push(formatDateLocal(curDate));
          curDate.setDate(curDate.getDate() + 1);
      }

      let dailyWages = {}; let dailyWorkingCounts = {}; let totalWage = 0; let workingEmployees = new Set();

      dateStrings.forEach(dStr => {
          dailyWages[dStr] = 0; dailyWorkingCounts[dStr] = 0;
          currentLoadedEmployees.forEach(emp => {
              let key = `${emp.hrCode}_${dStr}`;
              let data = expDataMap[key];
              let baseWage = parseFloat(emp.dailyWage) || 0;

              if (data && data.otherNote !== "ล้างข้อมูล") {
                  const stop = data.isStop; const cin = data.checkIn; const type = data.empType;
                  const wage = parseFloat(data.wage) || 0; 

                  if(!stop && cin) { workingEmployees.add(data.hrCode); dailyWorkingCounts[dStr]++; }

                  dailyWages[dStr] += wage; 
                  totalWage += wage;
              } else {
                  if (emp.empType === 'F/T') { dailyWages[dStr] += baseWage; totalWage += baseWage; }
              }
          });
      });

      document.getElementById('expWorkingCount').innerText = formatNumber(workingEmployees.size);
      document.getElementById('expTotalWage').innerText = formatNumber(totalWage);
      const pctElem = document.getElementById('expWagePercentDisplay');
      if(globalExpWeeklyTarget > 0) { pctElem.innerText = ((totalWage / globalExpWeeklyTarget) * 100).toFixed(2) + '%'; } else { pctElem.innerText = '0.00%'; }

      let headHtml = `<tr><th style="min-width: 180px;">พนักงาน | ตำแหน่ง</th>`;
      const daysTH = ['จันทร์', 'อังคาร', 'พุธ', 'พฤหัสบดี', 'ศุกร์', 'เสาร์', 'อาทิตย์'];
      curDate = new Date(expWeekStartDate); 
      
      for(let i=0; i<7; i++) {
          let dStr = dateStrings[i]; let dayWage = dailyWages[dStr] || 0; let dayCount = dailyWorkingCounts[dStr] || 0;
          
          let isHoliday = publicHolidays.includes(dStr);
          let holidayIcon = isHoliday ? ' <i class="bi bi-star-fill text-warning" title="วันนักขัตฤกษ์"></i>' : '';

          headHtml += `<th style="width: 12%; min-width: 110px;">
                          ${daysTH[i]}${holidayIcon}<br>
                          <span class="fw-normal" style="font-size:11px;">${curDate.getDate()} ${curDate.toLocaleDateString('th-TH',{month:'short'})}</span><br>
                          <span class="badge bg-success mt-1 fw-normal" style="font-size:11px;">฿ ${formatNumber(dayWage)}</span><br>
                          <span class="badge bg-info mt-1 fw-normal text-dark" style="font-size:11px;"><i class="bi bi-person-fill"></i> ${formatNumber(dayCount)} คน</span>
                       </th>`;
          curDate.setDate(curDate.getDate() + 1);
      }
      headHtml += `</tr>`;
      thead.innerHTML = headHtml;

      let bodyHtml = '';
      currentLoadedEmployees.forEach(emp => {
          bodyHtml += `<tr>
              <td>
                  <div class="emp-info-exp">
                      <span class="emp-name-exp">${emp.name}</span>
                      <span class="text-secondary" style="font-size: 10px;">รหัส: ${emp.hrCode || '-'}</span>
                      <span class="emp-pos-exp text-muted">${emp.position} <span class="badge bg-light text-dark border ms-1">${emp.empType}</span></span>
                  </div>
              </td>`;
          for(let i=0; i<7; i++) {
              let dStr = dateStrings[i]; let key = `${emp.hrCode}_${dStr}`; let cellData = expDataMap[key] || null;
              bodyHtml += `<td class="p-1">${buildCellCardHtml(emp, dStr, cellData)}</td>`;
          }
          bodyHtml += `</tr>`;
      });
      tbody.innerHTML = bodyHtml;
  }

  function buildCellCardHtml(emp, dateStr, data) {
      const onclickAttr = `onclick="openExpModal('${emp.hrCode}', '${dateStr}', '${emp.name}', '${emp.empType}', ${emp.dailyWage||0})"`;
      if(!data) return `<div class="shift-card shift-empty" ${onclickAttr}><i class="bi bi-plus-lg"></i> เพิ่มกะ</div>`;
      
      if(data.isStop) {
          let leaveTexts = [];
          if(data.leaveNote) leaveTexts.push(data.leaveNote);
          if(data.unpaidLeave) leaveTexts.push(data.unpaidLeave);
          if(data.otherNote && data.otherNote !== "ล้างข้อมูล") leaveTexts.push(data.otherNote);

          let reasonHtml = '';
          if(leaveTexts.length > 0) {
              reasonHtml = `<div class="mt-1"><span class="badge bg-danger text-white" style="font-size: 10px; white-space: normal; line-height: 1.2;">${leaveTexts.join(', ')}</span></div>`;
          }
          
          return `<div class="shift-card shift-stop" ${onclickAttr} style="flex-direction: column; justify-content: center; gap: 0;">
                      <div class="fw-bold"><i class="bi bi-dash-circle"></i> หยุด</div>
                      ${reasonHtml}
                  </div>`;
      }
      
      let isHoliday = publicHolidays.includes(dateStr);

      let otVal = parseFloat(data.ot) || 0;
      let leaveTexts = [];
      if(data.leaveNote) leaveTexts.push(data.leaveNote);
      if(data.unpaidLeave) leaveTexts.push(data.unpaidLeave);
      if(data.hourlyLeave) leaveTexts.push(`ลาชม. ${data.hourlyLeave}`);
      if(data.useAccumulatedHours) leaveTexts.push(`ใช้ชม.สะสม ${data.useAccumulatedHours}ชม.`);
      if(data.otherNote && data.otherNote !== "ล้างข้อมูล") leaveTexts.push(data.otherNote);

      let badges = [];
      if (otVal > 0) badges.push(`<span class="badge bg-primary">OT ${data.ot}</span>`);
      if (leaveTexts.length > 0) badges.push(`<span class="badge bg-warning text-dark" style="white-space: normal; line-height: 1.2;">${leaveTexts.join(', ')}</span>`);
      
      if (isHoliday && data.checkIn) {
          badges.push(`<span class="badge bg-danger text-white" style="white-space: normal; line-height: 1.2;">นักขัตฤกษ์ ⭐</span>`);
      }

      let topHtml = '';
      if (badges.length > 0) topHtml = badges.join(' ');
      else {
          let shiftName = emp.empType === 'F/T' ? 'F/T' : (emp.empType === 'DAY9' ? 'DAY9' : 'DAY');
          topHtml = `<span class="badge bg-warning text-dark">${shiftName}</span>`;
      }

      let timeDisplay = data.checkIn ? `<div class="shift-time mt-1">${data.checkIn} - ${data.checkOut||'?'}</div>` : '';
      return `<div class="shift-card" ${onclickAttr}><div class="mb-1">${topHtml}</div>${timeDisplay}</div>`;
  }

  function openExpModal(hrCode, dateStr, empName, empType, wageRate) {

      document.getElementById('modalOTAccum').value = data.otAccumulated || "0";

      document.getElementById('modalLeave2').value = data.unpaidLeave || "";
      document.getElementById('modalHrLeave').value = data.hourlyLeave || "";

  function toggleModalFields() {
      const stop = document.getElementById('modalIsStop').checked;
      const l1 = document.getElementById('modalLeave1').value;
      const l2 = document.getElementById('modalLeave2').value;
      
      let disableAll = stop || (l1 !== "") || (l2 !== "");
      
      ['modalBreakDur','modalOT','modalOTAccum','modalHrLeave','modalUseAccum'].forEach(id => {
          let el = document.getElementById(id);
          el.disabled = disableAll;
          if(disableAll) { 
              if(el.tagName==='SELECT') el.value = el.options[0].value; else el.value=''; 
          } else {
              if(id === 'modalBreakDur' && el.value === '') el.value = '60';
          }
      });

      ['modalCheckIn','modalCheckOut','modalBreakStart'].forEach(type => {

  function calcModalEnd() {
      const pt = document.getElementById('modalEmpType').value === 'P/T';
      if(pt) return; 

      const ci = document.getElementById('modalCheckIn').value;
      if(!ci) return;
      
      const empType = document.getElementById('modalEmpType').value;
      const baseMin = (empType === 'DAY9') ? 540 : 480;
      
      const [h,m] = ci.split(':').map(Number);
      let total = h*60 + m + baseMin; 
      total += (parseInt(document.getElementById('modalBreakDur').value)||0);
      total += (parseFloat(document.getElementById('modalOT').value)||0) * 60;
      total += (parseFloat(document.getElementById('modalOTAccum').value)||0) * 60;
      
      let eh = Math.floor(total/60); 
      let em = total%60;
      if (eh >= 24) {
          if (eh === 24 && em === 0) {} else { eh = eh % 24; }
      }

      let hrStr = String(eh).padStart(2,'0'); let minStr = String(em).padStart(2,'0');
      document.getElementById('modalCheckOut').value = `${hrStr}:${minStr}`;
      
      ensureSelectOption(document.getElementById('modalCheckOutHr'), hrStr);
      ensureSelectOption(document.getElementById('modalCheckOutMin'), minStr);
      document.getElementById('modalCheckOutHr').value = hrStr;
      document.getElementById('modalCheckOutMin').value = minStr;
  }

  function saveModalData() {
      const hrCode = document.getElementById('modalHrCode').value;
      const dateStr = document.getElementById('modalDate').value;
      const empType = document.getElementById('modalEmpType').value;
      const rate = parseFloat(document.getElementById('modalWageRate').value) || 0;
      
      const emp = currentLoadedEmployees.find(e => e.hrCode === hrCode);
      if(!emp) return;

      const stop = document.getElementById('modalIsStop').checked;
      const ci = document.getElementById('modalCheckIn').value;
      const co = document.getElementById('modalCheckOut').value;
      const brDur = document.getElementById('modalBreakDur').value;
      const bs = document.getElementById('modalBreakStart').value;
      const ot = document.getElementById('modalOT').value;
      const ota = document.getElementById('modalOTAccum').value;
      const l1 = document.getElementById('modalLeave1').value;
      const l2 = document.getElementById('modalLeave2').value;
      const hl = document.getElementById('modalHrLeave').value;
      const ua = document.getElementById('modalUseAccum').value;
      const noteInput = document.getElementById('modalOtherNote').value.trim();
      
      const finalStop = stop || (l1 !== "") || (l2 !== "");

      // บังคับว่าถ้ากดกำหนดเป็นหยุด ต้องเลือกเหตุผล
      if (stop && l1 === "" && l2 === "") {
          Swal.fire('เตือน', 'กรุณาระบุประเภทการลา หรือ เลือก "วันหยุดธรรมดา" ในช่องหยุดไม่รับค่าแรง', 'warning');
          return;
      }

      if(!finalStop && ci) {
          if(!brDur) { Swal.fire('เตือน','กรุณาระบุเวลาเบรค','warning'); return; }
          if(brDur !== "0" && !bs) { Swal.fire('เตือน','กรุณาระบุเวลาเริ่มเบรค','warning'); return; }
      }

      let brRange = "";
      if(bs && brDur && brDur !== "0") {
          const [h,m] = bs.split(':').map(Number);
          const t = h*60 + m + parseInt(brDur);
          const eh = Math.floor(t/60)%24; const em = t%60;
          brRange = `${bs}-${String(eh).padStart(2,'0')}:${String(em).padStart(2,'0')}`;
      }
      
      let wage = 0; let isCleared = false;
      if(!ci && !co && !finalStop && !l1 && !l2 && (!ota || ota==="0") && !hl && !ua && !noteInput) { isCleared = true; }

      if (isCleared) {
          if (empType === 'F/T') wage = rate; else wage = 0;
      } else {
          if (l2 !== "") { 
              if (l2 === "วันหยุดธรรมดา" && empType === 'F/T') {
                  wage = rate; 
              } else {
                  wage = 0; 
              }
          } 
          else if (l1 !== "") { 
              if (empType === 'P/T') { wage = 0; } else { wage = rate; } 
          } 
          else {
              if (empType === 'F/T') {
                  wage = rate; 
                  if (hl !== "") {
                      let h = hl==='ครึ่งวัน'?4:hl==='เต็มวัน'?8:parseInt(hl)||0;
                      if(rate>0) wage = rate - (rate/8)*h;
                  }
              } else if (empType === 'DAY' || empType === 'DAY9') {
                  let baseHours = (empType === 'DAY9') ? 9 : 8;
                  if (!finalStop && ci) {
                      wage = rate; 
                      if (hl !== "") {
                          let h = hl==='ครึ่งวัน'? (baseHours/2) : hl==='เต็มวัน'? baseHours : parseInt(hl)||0;
                          if(rate>0) wage = rate - (rate/baseHours)*h;
                      }
                  } else { wage = 0; }
              } else if (empType === 'P/T') {
                  if (!finalStop && ci && co) {
                      const [h1,m1] = ci.split(':').map(Number); const [h2,m2] = co.split(':').map(Number);
                      let t1 = h1*60+m1; let t2 = h2*60+m2; if(t2<t1) t2+=1440;
                      let dur = (t2-t1) - (parseInt(brDur)||0);
                      if (dur > 0) wage = (dur/60)*rate;
                  } else { wage = 0; }
              }
          }
      }
      wage = parseFloat(wage.toFixed(2));
      
      let brTxt = (brDur==="0")?"ไม่เบรค":brDur?`${parseInt(brDur)/60} ชม.`:"";
      const key = `${hrCode}_${dateStr}`;

      const payload = {
          workDate: dateStr, hrCode: emp.hrCode, name: emp.name, branch: emp.branch, position: emp.position, empType: emp.empType,
          checkIn: ci, checkOut: co, breakTime: brTxt, breakTimeRange: brRange, ot: ot, otAccumulated: ota, wage: wage, isStop: finalStop,
          leaveNote: l1, unpaidLeave: l2, hourlyLeave: hl, otherNote: isCleared ? "ล้างข้อมูล" : noteInput, workStation: "", useAccumulatedHours: ua
      };

      expDataMap[key] = payload;
      expModifiedKeys.add(key); 

      cellEditModalInstance.hide();
      renderExpGrid(); 
  }

              let nameDisplay = `<div>${r.name}${holidayHtml}</div><div class="text-secondary" style="font-size: 10px;">รหัส: ${hrCodeText}</div>`;
              let html = `<td style="text-align:left;">${pos}</td><td style="padding:0;">${sel}</td><td style="text-align:left;">${nameDisplay}</td>`;
              
              for(let h=8; h<=24; h++) {
                  let ws=parsedIn.val, we=parsedOut.val, ss=h, se=h+1;
                  let isW = false;
                  
                  if(Math.max(ss, ws) < Math.min(se, we)) {
                      isW=true;
                      if(bs && be && Math.max(ss, bs.val) < Math.min(se, be.val)) {
                          let overlapW = Math.min(se, we) - Math.max(ss, ws);
                          let overlapB = Math.max(0, Math.min(Math.min(se, we), be.val) - Math.max(Math.max(ss, ws), bs.val));
                          if(overlapB >= overlapW && overlapW > 0) isW = false;
                      }
                  }
                  if(isW) { stats[h].total++; if(!stats[h].pos[pos]) stats[h].pos[pos]=0; stats[h].pos[pos]++; }
                  let sty = getScheduleCellStyle(h, ws, we, bs?bs.val:-1, be?be.val:-1);
                  html += `<td class="schedule-cell" style="${sty}"></td>`;
              }
              tr.innerHTML = html; tb.appendChild(tr);
          });

          document.getElementById('scheduleSummaryPanel').style.display='flex';
          document.getElementById('schedTotalWage').innerText = formatNumber(totalW);
          updateWagePercentage('schedTotalWage', globalScheduleDailyTarget, 'schedWagePercent');

          let ft = '<tr style="background-color:#e9ecef;font-weight:bold;border-top:2px solid #dee2e6;"><td colspan="3" class="text-end">รวมพนักงาน (คน)</td>';
          for(let h=8; h<=24; h++) ft+=`<td class="text-center">${stats[h].total||''}</td>`;
          ft+='</tr>';
          allPos.forEach(p => {
              ft+=`<tr><td colspan="3" class="text-end text-muted small" style="font-size:11px;">${p}</td>`;
              for(let h=8; h<=24; h++) { let c = stats[h].pos[p]||0; ft+=`<td class="text-center small text-muted" style="${c>0?'background-color:#f0fff4;':''}">${c||''}</td>`; }
              ft+='</tr>';
          });
          tf.innerHTML = ft;
      } catch(e) {
          tb.innerHTML = `<tr><td colspan="20" class="text-danger text-center py-4">แสดงผลขัดข้อง: ${e.message}</td></tr>`;
      }
  }
